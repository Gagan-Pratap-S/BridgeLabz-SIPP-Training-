CLASS Savings EXTENDS Account
    PROPERTIES
        withdrawalLimit

    METHODS
        withdraw(amount)
            IF amount > balance OR amount > withdrawalLimit THEN
                PRINT "Withdrawal Denied"
            ELSE
                balance -= amount

CLASS Current EXTENDS Account
    METHODS
        withdraw(amount)
            IF amount > balance THEN
                PRINT "Overdraft Not Allowed"
            ELSE

Q3: Flight price Analyzer:

CLASS CityZone
    PROPERTIES
        zoneId
        crimeRate
        neighbors (List<CityZone>)

METHOD findHighRiskZones(graph)
    INIT visited set
    INIT result list

    FOR each zone IN graph
        IF zone NOT IN visited AND zone.crimeRate > threshold
            DFS(zone, visited, result)

METHOD DFS(zone, visited, result)
    MARK zone AS visited
    ADD zone TO result
    FOR each neighbor IN zone.neighbors
        IF neighbor NOT IN visited AND neighbor.crimeRate > threshold
            DFS(neighbor, visited, result)
			
Q4: College Course Registration

CLASS Course
    PROPERTIES
        courseId
        courseName

CLASS Student
    PROPERTIES
        studentId
        name

CLASS CourseRegistration
    PROPERTIES
        map<Course, List<Student>> registrations

    METHODS
        registerStudent(course, student)
            IF course NOT IN registrations
                registrations[course] = new List
            registrations[course].add(student)

        displayStudentsPerCourse()
            FOR each course IN registrations
                PRINT course.courseName
                FOR each student IN registrations[course]
                    PRINT student.name
